cmake_minimum_required(VERSION 3.8)
project(v87_nav2_observation_window)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(plan_env REQUIRED)
find_package(scan_planner REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(nav2_util REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
add_executable(observation_window_probe observation_window_probe.cpp)
target_compile_options(observation_window_probe PRIVATE -fno-access-control)
ament_target_dependencies(observation_window_probe plan_env nav2_costmap_2d nav2_util)
target_link_libraries(observation_window_probe Eigen3::Eigen Threads::Threads)
add_executable(native_window_probe native_window_probe.cpp)
target_compile_options(native_window_probe PRIVATE -fno-access-control)
ament_target_dependencies(native_window_probe plan_env)
target_link_libraries(native_window_probe Eigen3::Eigen OpenSSL::Crypto Threads::Threads)

add_executable(transient_window_probe transient_window_probe.cpp)
target_compile_options(transient_window_probe PRIVATE -fno-access-control)
ament_target_dependencies(transient_window_probe plan_env)
target_link_libraries(transient_window_probe Eigen3::Eigen OpenSSL::Crypto Threads::Threads)

add_executable(safety_producer_probe safety_producer_probe.cpp)
target_compile_options(safety_producer_probe PRIVATE -fno-access-control)
ament_target_dependencies(safety_producer_probe scan_planner plan_env tf2 tf2_geometry_msgs)
target_link_libraries(safety_producer_probe Eigen3::Eigen OpenSSL::Crypto Threads::Threads)
