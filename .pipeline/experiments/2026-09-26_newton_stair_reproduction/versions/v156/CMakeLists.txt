cmake_minimum_required(VERSION 3.16)
project(stopped_corner_query)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(plan_env REQUIRED)
find_package(scan_planner REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
add_executable(stopped_query_probe probe.cpp)
target_compile_options(stopped_query_probe PRIVATE -fno-access-control)
target_include_directories(stopped_query_probe BEFORE PRIVATE ${VERSIONS_ROOT})
ament_target_dependencies(stopped_query_probe rclcpp plan_env scan_planner)
target_link_libraries(stopped_query_probe OpenSSL::Crypto Threads::Threads)
