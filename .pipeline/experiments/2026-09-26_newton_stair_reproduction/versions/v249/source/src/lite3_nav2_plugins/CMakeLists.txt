cmake_minimum_required(VERSION 3.8)
project(lite3_nav2_plugins)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
set(dependencies plan_env lite3_nav2_interfaces nav2_core nav2_costmap_2d nav2_behavior_tree pluginlib rclcpp
  rclcpp_lifecycle nav_msgs geometry_msgs std_srvs pct_planner tf2_ros)
foreach(dep ${dependencies})
  find_package(${dep} REQUIRED)
endforeach()
find_package(Eigen3 REQUIRED)
find_package(GTSAM REQUIRED)
add_library(pct_route_planner SHARED src/pct_route_planner.cpp)
target_include_directories(pct_route_planner PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(pct_route_planner pct_planner plan_env)
target_link_libraries(pct_route_planner Eigen3::Eigen gtsam)
add_library(lite3_nav2_plugins SHARED src/pct_planner.cpp)
target_link_libraries(lite3_nav2_plugins pct_route_planner)
ament_target_dependencies(lite3_nav2_plugins ${dependencies})
pluginlib_export_plugin_description_file(nav2_core plugins.xml)
add_library(lite3_nav2_route_bt SHARED src/recovery_bt.cpp)
target_include_directories(lite3_nav2_route_bt PRIVATE include)
target_compile_definitions(lite3_nav2_route_bt PRIVATE BT_PLUGIN_EXPORT)
ament_target_dependencies(lite3_nav2_route_bt ${dependencies})
target_link_libraries(lite3_nav2_route_bt /opt/ros/foxy/lib/libnav2_follow_path_action_bt_node.so)
install(TARGETS lite3_nav2_plugins pct_route_planner lite3_nav2_route_bt LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(bspline_opt REQUIRED)
  ament_add_gtest(test_physical_support test/test_physical_support.cpp TIMEOUT 30)
  target_compile_options(test_physical_support PRIVATE -fno-access-control)
  ament_target_dependencies(test_physical_support plan_env pct_planner)
  target_link_libraries(test_physical_support pct_route_planner Eigen3::Eigen)
  ament_add_gtest(test_environmental_cost test/test_environmental_cost.cpp TIMEOUT 90)
  target_compile_options(test_environmental_cost PRIVATE -fno-access-control)
  ament_target_dependencies(test_environmental_cost plan_env bspline_opt)
  target_link_libraries(test_environmental_cost pct_route_planner)
  find_package(yaml_cpp_vendor REQUIRED)
  ament_add_gtest(test_route_policy test/test_route_policy.cpp TIMEOUT 10)
  ament_target_dependencies(test_route_policy ${dependencies})
  target_include_directories(test_route_policy PRIVATE include)
  target_link_libraries(test_route_policy Eigen3::Eigen yaml-cpp)
  ament_add_gtest(test_shared_traversability test/test_shared_traversability.cpp TIMEOUT 45)
  ament_target_dependencies(test_shared_traversability ${dependencies})
  target_include_directories(test_shared_traversability PRIVATE include)
  target_link_libraries(test_shared_traversability pct_route_planner Eigen3::Eigen)
  add_executable(pct_route_probe test/pct_route_probe.cpp)
  target_link_libraries(pct_route_probe pct_route_planner)
  ament_add_gtest(test_direct_pct test/test_direct_pct.cpp TIMEOUT 30)
  ament_target_dependencies(test_direct_pct ${dependencies})
  target_link_libraries(test_direct_pct pct_route_planner)
  ament_add_gtest(test_progress_recovery test/test_progress_recovery.cpp TIMEOUT 30)
  ament_target_dependencies(test_progress_recovery ${dependencies})
  target_link_libraries(test_progress_recovery /opt/ros/foxy/lib/libnav2_recovery_node_bt_node.so)
endif()
ament_export_include_directories(include)
ament_package()
