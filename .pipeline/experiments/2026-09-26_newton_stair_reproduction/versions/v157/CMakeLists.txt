cmake_minimum_required(VERSION 3.16)
project(body_aligned_search_regression)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(plan_env REQUIRED)
find_package(scan_planner REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
add_executable(stopped_recovery_probe ../v156/probe.cpp)
add_executable(coordinate_regression coordinate_regression.cpp)
foreach(target stopped_recovery_probe coordinate_regression)
  target_compile_options(${target} PRIVATE -fno-access-control)
  target_include_directories(${target} BEFORE PRIVATE ${VERSIONS_ROOT})
  ament_target_dependencies(${target} rclcpp plan_env scan_planner)
  target_link_libraries(${target} OpenSSL::Crypto Threads::Threads)
endforeach()
