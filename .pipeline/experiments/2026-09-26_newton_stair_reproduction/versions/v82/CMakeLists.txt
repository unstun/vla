cmake_minimum_required(VERSION 3.8)
project(v82_conditional_backlog_contract)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(plan_env REQUIRED)
find_package(bspline_opt REQUIRED)
foreach(target map_probe backlog_probe async_contract_probe)
  add_executable(${target} ${target}.cpp)
  target_include_directories(${target} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  target_compile_options(${target} PRIVATE -fno-access-control)
  ament_target_dependencies(${target} plan_env bspline_opt)
  target_link_libraries(${target} Eigen3::Eigen OpenSSL::Crypto)
endforeach()
