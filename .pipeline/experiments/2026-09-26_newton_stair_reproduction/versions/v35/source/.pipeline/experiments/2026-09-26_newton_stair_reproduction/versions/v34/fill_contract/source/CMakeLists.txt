cmake_minimum_required(VERSION 3.8)
project(fill_contract)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
set(dependencies ament_index_cpp bspline_opt geometry_msgs nav_msgs path_searching plan_env rclcpp rosgraph_msgs scan_planner_msgs sensor_msgs std_msgs tf2 tf2_geometry_msgs tf2_ros traj_utils visualization_msgs)
foreach(dependency ${dependencies})
 find_package(${dependency} REQUIRED)
endforeach()
# Header part: the fill and step score of the tree under test only.
add_executable(fill_contract src/fill_contract.cpp)
target_include_directories(fill_contract PRIVATE /workspace/src/plan_env/include)
# Node part: the same manager objects as the regression probe, compiled from the tree under test.
add_library(native_manager OBJECT /workspace/src/plan_manage/src/scan_replan_fsm.cpp /workspace/src/plan_manage/src/planner_manager.cpp)
target_include_directories(native_manager PRIVATE /workspace/src/plan_manage/include)
ament_target_dependencies(native_manager ${dependencies})
target_link_libraries(native_manager Eigen3::Eigen)
add_executable(fill_order_contract src/fill_order_contract.cpp $<TARGET_OBJECTS:native_manager>)
target_include_directories(fill_order_contract PRIVATE /workspace/src/plan_manage/include)
target_compile_options(fill_order_contract PRIVATE -fno-access-control)
ament_target_dependencies(fill_order_contract ${dependencies})
target_link_libraries(fill_order_contract Eigen3::Eigen)
ament_package()
