cmake_minimum_required(VERSION 3.8)
project(native_constraint_contract)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(rclcpp REQUIRED)
add_executable(contract native_tests.cpp)
target_include_directories(contract PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/generated)
target_compile_options(contract PRIVATE -D_GLIBCXX_ASSERTIONS -fsanitize=undefined -g -O1)
target_link_options(contract PRIVATE -fsanitize=undefined)
ament_target_dependencies(contract rclcpp)
target_link_libraries(contract Eigen3::Eigen)
