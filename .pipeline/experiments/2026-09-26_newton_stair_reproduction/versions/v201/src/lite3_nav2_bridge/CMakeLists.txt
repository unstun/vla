cmake_minimum_required(VERSION 3.8)
project(lite3_nav2_bridge)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
ament_python_install_package(${PROJECT_NAME})
install(PROGRAMS scripts/mission_bridge DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(behaviortree_cpp_v3 REQUIRED)
  ament_add_gtest(test_terminal_tree test/test_terminal_tree.cpp TIMEOUT 10)
  ament_target_dependencies(test_terminal_tree behaviortree_cpp_v3)
  target_compile_definitions(test_terminal_tree PRIVATE NAVIGATION_TREE="${CMAKE_CURRENT_SOURCE_DIR}/config/navigate.xml")
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(test_mission_bridge test/test_mission_bridge.py TIMEOUT 60)
endif()
ament_package()
