cmake_minimum_required(VERSION 3.16)
project(v240_publication_contract)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(scan_planner REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
add_executable(publication_contract publication_contract.cpp)
target_compile_options(publication_contract PRIVATE -fno-access-control)
target_include_directories(publication_contract BEFORE PRIVATE ${VERSIONS_ROOT})
ament_target_dependencies(publication_contract rclcpp scan_planner)
target_link_libraries(publication_contract OpenSSL::Crypto Threads::Threads)
