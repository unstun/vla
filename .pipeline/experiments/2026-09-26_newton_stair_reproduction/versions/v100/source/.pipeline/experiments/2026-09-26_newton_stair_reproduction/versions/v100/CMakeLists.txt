cmake_minimum_required(VERSION 3.8)
project(v100_clock_collision_probe)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(rosgraph_msgs REQUIRED)
find_package(Threads REQUIRED)
add_executable(clock_lane_probe clock_lane_probe.cpp)
ament_target_dependencies(clock_lane_probe rclcpp nav_msgs rosgraph_msgs)
target_link_libraries(clock_lane_probe Threads::Threads)
option(V100_CLOCK_ONLY "Build only the independent ROS clock scheduling probe" OFF)
if(V100_CLOCK_ONLY)
  return()
endif()
find_package(plan_env REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(OpenSSL REQUIRED)
add_executable(clock_collision_probe clock_collision_probe.cpp)
target_compile_options(clock_collision_probe PRIVATE -fno-access-control)
ament_target_dependencies(clock_collision_probe plan_env)
target_link_libraries(clock_collision_probe Eigen3::Eigen OpenSSL::Crypto Threads::Threads)
