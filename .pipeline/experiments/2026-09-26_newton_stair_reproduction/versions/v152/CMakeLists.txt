cmake_minimum_required(VERSION 3.16)
project(recorded_replacement)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(plan_env REQUIRED)
find_package(scan_planner REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
add_executable(replacement_probe probe.cpp)
target_compile_options(replacement_probe PRIVATE -fno-access-control)
target_include_directories(replacement_probe BEFORE PRIVATE ${VERSIONS_ROOT})
ament_target_dependencies(replacement_probe rclcpp plan_env scan_planner)
target_link_libraries(replacement_probe OpenSSL::Crypto Threads::Threads)
