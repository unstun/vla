cmake_minimum_required(VERSION 3.16)
project(saved_output_heading_audit)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
find_package(rclcpp REQUIRED)
find_package(scan_planner REQUIRED)
find_package(plan_env REQUIRED)
find_package(nav2_core REQUIRED)
find_package(pluginlib REQUIRED)
find_package(lite3_nav2_plugins REQUIRED)
find_package(lite3_nav2_interfaces REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
add_executable(runtime_curve_probe runtime_curve_probe.cpp)
target_compile_options(runtime_curve_probe PRIVATE -fno-access-control)
target_include_directories(runtime_curve_probe BEFORE PRIVATE ${VERSIONS_ROOT})
ament_target_dependencies(runtime_curve_probe rclcpp scan_planner lite3_nav2_plugins lite3_nav2_interfaces rclcpp_lifecycle nav2_core pluginlib)
add_executable(endpoint_witness ${AUDIT_ROOT}/endpoint_witness.cpp)
ament_target_dependencies(endpoint_witness plan_env)

target_link_libraries(runtime_curve_probe OpenSSL::Crypto Threads::Threads)

target_compile_options(endpoint_witness PRIVATE -fno-access-control)
