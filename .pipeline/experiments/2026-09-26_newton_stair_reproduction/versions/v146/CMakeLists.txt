cmake_minimum_required(VERSION 3.16)
project(sampled_rebound_probe)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(plan_env REQUIRED)
find_package(scan_planner REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
add_executable(envelope_probe probe.cpp)
target_compile_options(envelope_probe PRIVATE -fno-access-control)
target_include_directories(envelope_probe BEFORE PRIVATE ${VERSIONS_ROOT})
ament_target_dependencies(envelope_probe rclcpp plan_env scan_planner)
target_link_libraries(envelope_probe OpenSSL::Crypto Threads::Threads)
