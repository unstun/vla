set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)

set(dependencies ament_index_cpp bspline_opt geometry_msgs nav_msgs path_searching plan_env rclcpp rosgraph_msgs scan_planner_msgs sensor_msgs std_msgs tf2 tf2_geometry_msgs tf2_ros traj_utils visualization_msgs)
foreach(dependency ${dependencies})
  find_package(${dependency} REQUIRED)
endforeach()
add_executable(periodic_probe periodic_probe.cpp
  /workspace/src/plan_manage/src/scan_replan_fsm.cpp
  /workspace/src/plan_manage/src/planner_manager.cpp)
target_include_directories(periodic_probe PRIVATE /workspace/src/plan_manage/include)
# Same test-only native access pattern as the existing corridor_probe.
target_compile_options(periodic_probe PRIVATE -fno-access-control)
if(LEGACY_PARENT)
  target_compile_definitions(periodic_probe PRIVATE V35_LEGACY_PARENT)
endif()
ament_target_dependencies(periodic_probe ${dependencies})
