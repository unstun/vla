cmake_minimum_required(VERSION 3.16)
project(nested_sampling_regression)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(plan_env REQUIRED)
find_package(scan_planner REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
add_executable(sample_regression sample_regression.cpp)
add_executable(safety_pair_candidate ../v150/probe.cpp)
foreach(target sample_regression safety_pair_candidate)
  target_compile_options(${target} PRIVATE -fno-access-control)
  target_include_directories(${target} BEFORE PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/src/bspline_opt/include ${VERSIONS_ROOT})
  ament_target_dependencies(${target} rclcpp plan_env scan_planner)
  target_link_libraries(${target} OpenSSL::Crypto Threads::Threads)
endforeach()
target_compile_definitions(sample_regression PRIVATE
  V147_CHECK_HEADER="/task/recovery_v147/src/bspline_opt/include/bspline_opt/full_trajectory_collision.h")
