cmake_minimum_required(VERSION 3.8)
project(v78_map_processing_profile)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(plan_env REQUIRED)
foreach(target baseline_map_probe instrumented_map_probe)
  add_executable(${target} map_probe.cpp)
  target_include_directories(${target} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} /generated)
  target_compile_options(${target} PRIVATE -fno-access-control)
  ament_target_dependencies(${target} plan_env)
  target_link_libraries(${target} Eigen3::Eigen OpenSSL::Crypto)
endforeach()
target_sources(instrumented_map_probe PRIVATE /generated/instrumented_grid_map.cpp)
