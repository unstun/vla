cmake_minimum_required(VERSION 3.8)
project(waiver_contract)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
set(dependencies ament_index_cpp bspline_opt geometry_msgs nav_msgs path_searching plan_env rclcpp rosgraph_msgs scan_planner_msgs sensor_msgs std_msgs tf2 tf2_geometry_msgs tf2_ros traj_utils visualization_msgs)
foreach(dependency ${dependencies})
 find_package(${dependency} REQUIRED)
endforeach()
# Same manager objects as the regression probe, compiled from the tree under test.
add_library(native_manager OBJECT /workspace/src/plan_manage/src/scan_replan_fsm.cpp /workspace/src/plan_manage/src/planner_manager.cpp)
target_include_directories(native_manager PRIVATE /workspace/src/plan_manage/include)
ament_target_dependencies(native_manager ${dependencies})
target_link_libraries(native_manager Eigen3::Eigen)
add_executable(waiver_contract src/waiver_contract.cpp $<TARGET_OBJECTS:native_manager>)
target_include_directories(waiver_contract PRIVATE /workspace/src/plan_manage/include)
target_compile_options(waiver_contract PRIVATE -fno-access-control)
ament_target_dependencies(waiver_contract ${dependencies})
target_link_libraries(waiver_contract Eigen3::Eigen)
ament_package()
