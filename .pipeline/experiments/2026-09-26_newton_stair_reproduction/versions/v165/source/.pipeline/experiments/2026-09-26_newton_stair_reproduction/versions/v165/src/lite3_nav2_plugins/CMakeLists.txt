cmake_minimum_required(VERSION 3.8)
project(lite3_nav2_plugins)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
set(dependencies nav2_core nav2_costmap_2d nav2_behavior_tree pluginlib rclcpp
  rclcpp_lifecycle nav_msgs geometry_msgs std_srvs lite3_nav2_interfaces tf2_ros)
foreach(dep ${dependencies})
  find_package(${dep} REQUIRED)
endforeach()
add_library(lite3_nav2_plugins SHARED src/pct_planner.cpp src/terrain_layer.cpp)
ament_target_dependencies(lite3_nav2_plugins ${dependencies})
pluginlib_export_plugin_description_file(nav2_core plugins.xml)
pluginlib_export_plugin_description_file(nav2_costmap_2d layers.xml)
add_library(lite3_bind_route_bt_node SHARED src/bind_route.cpp)
target_compile_definitions(lite3_bind_route_bt_node PRIVATE BT_PLUGIN_EXPORT)
ament_target_dependencies(lite3_bind_route_bt_node ${dependencies})
install(TARGETS lite3_nav2_plugins lite3_bind_route_bt_node LIBRARY DESTINATION lib)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_terrain_layer test/test_terrain_layer.cpp TIMEOUT 30)
  ament_target_dependencies(test_terrain_layer ${dependencies})
  ament_add_gtest(test_reference_cache test/test_reference_cache.cpp TIMEOUT 30)
  ament_target_dependencies(test_reference_cache ${dependencies})
endif()
ament_package()
