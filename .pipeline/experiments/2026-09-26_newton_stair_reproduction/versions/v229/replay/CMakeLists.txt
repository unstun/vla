cmake_minimum_required(VERSION 3.16)
project(stair_collision_readonly_probe)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(plan_env REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
add_executable(collision_probe collision_probe.cpp)
target_compile_options(collision_probe PRIVATE -fno-access-control)
target_include_directories(collision_probe BEFORE PRIVATE ${VERSIONS_ROOT})
ament_target_dependencies(collision_probe rclcpp plan_env)
target_link_libraries(collision_probe OpenSSL::Crypto Threads::Threads)
