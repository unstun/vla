cmake_minimum_required(VERSION 3.8)
project(scan_nav2_controller)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(Threads REQUIRED)
set(dependencies lite3_nav2_interfaces rcl_yaml_param_parser scan_planner nav2_core pluginlib rclcpp rclcpp_lifecycle bspline_opt
  scan_planner_msgs geometry_msgs nav_msgs std_msgs tf2 tf2_geometry_msgs)
foreach(dependency ${dependencies})
  find_package(${dependency} REQUIRED)
endforeach()
add_library(scan_nav2_controller SHARED src/controller.cpp src/local_motion.cpp)
target_include_directories(scan_nav2_controller PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(scan_nav2_controller ${dependencies})
target_link_libraries(scan_nav2_controller Eigen3::Eigen Threads::Threads)
pluginlib_export_plugin_description_file(nav2_core plugins.xml)
install(TARGETS scan_nav2_controller EXPORT export_scan_nav2_controller
  ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
ament_export_targets(export_scan_nav2_controller HAS_LIBRARY_TARGET)
ament_export_dependencies(${dependencies})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_map_shell test/test_map_shell.cpp TIMEOUT 12)
  target_link_libraries(test_map_shell scan_nav2_controller)
  ament_target_dependencies(test_map_shell ${dependencies})
  ament_add_gtest(test_static_support test/test_static_support.cpp TIMEOUT 15)
  target_compile_options(test_static_support PRIVATE -fno-access-control)
  target_link_libraries(test_static_support scan_nav2_controller)
  ament_target_dependencies(test_static_support ${dependencies})
  ament_add_gtest(test_motion_exchange test/test_motion_exchange.cpp TIMEOUT 10)
  target_link_libraries(test_motion_exchange scan_nav2_controller)
  ament_target_dependencies(test_motion_exchange ${dependencies})
  ament_add_gtest(test_terminal_completion test/test_terminal_completion.cpp TIMEOUT 10)
  target_link_libraries(test_terminal_completion scan_nav2_controller)
  ament_target_dependencies(test_terminal_completion ${dependencies})
  ament_add_gtest(test_controller test/test_controller.cpp TIMEOUT 60)
  target_compile_options(test_controller PRIVATE -fno-access-control)
  target_link_libraries(test_controller scan_nav2_controller)
  target_compile_definitions(test_controller PRIVATE TEST_NATIVE_CONFIG="${CMAKE_CURRENT_SOURCE_DIR}/../plan_manage/config/planner.yaml")
  ament_target_dependencies(test_controller ${dependencies})
  ament_add_gtest(test_tracking_parity test/test_tracking_parity.cpp TIMEOUT 30)
  target_compile_options(test_tracking_parity PRIVATE -fno-access-control)
  target_link_libraries(test_tracking_parity scan_nav2_controller)
  ament_target_dependencies(test_tracking_parity ${dependencies})
  ament_add_gtest(test_executor_shutdown test/test_executor_shutdown.cpp TIMEOUT 10)
  target_link_libraries(test_executor_shutdown scan_nav2_controller)
  ament_target_dependencies(test_executor_shutdown ${dependencies})
endif()
ament_package()
