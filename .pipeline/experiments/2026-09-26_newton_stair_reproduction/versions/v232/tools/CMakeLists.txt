cmake_minimum_required(VERSION 3.16)
project(environment_route_probe)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(plan_env REQUIRED)
find_package(lite3_nav2_plugins REQUIRED)
find_package(pct_planner REQUIRED)
find_package(bspline_opt REQUIRED)
find_package(GTSAM REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
find_package(nav2_core REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
find_package(lite3_nav2_interfaces REQUIRED)
add_executable(field_probe field_probe.cpp)
target_compile_options(field_probe PRIVATE -fno-access-control)
target_include_directories(field_probe BEFORE PRIVATE ${VERSIONS_ROOT})
ament_target_dependencies(field_probe rclcpp plan_env pct_planner lite3_nav2_plugins bspline_opt)
target_link_libraries(field_probe /work/install/lite3_nav2_plugins/lib/libpct_route_planner.so gtsam OpenSSL::Crypto Threads::Threads)
add_executable(admission_probe admission_probe.cpp)

target_link_libraries(admission_probe OpenSSL::Crypto Threads::Threads)
