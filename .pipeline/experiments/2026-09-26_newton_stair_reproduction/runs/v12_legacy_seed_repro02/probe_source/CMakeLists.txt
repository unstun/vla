cmake_minimum_required(VERSION 3.8)
project(frontier_probe)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
set(dependencies ament_index_cpp bspline_opt geometry_msgs nav_msgs path_searching plan_env rclcpp rosgraph_msgs scan_planner_msgs sensor_msgs std_msgs tf2 tf2_geometry_msgs tf2_ros traj_utils visualization_msgs)
foreach(dependency ${dependencies})
 find_package(${dependency} REQUIRED)
endforeach()
add_executable(frontier_probe src/frontier_probe.cpp /workspace/src/plan_manage/src/scan_replan_fsm.cpp /workspace/src/plan_manage/src/planner_manager.cpp)
target_include_directories(frontier_probe PRIVATE /workspace/src/plan_manage/include)
target_compile_options(frontier_probe PRIVATE -fno-access-control -D_GLIBCXX_ASSERTIONS)
ament_target_dependencies(frontier_probe ${dependencies})
target_link_libraries(frontier_probe Eigen3::Eigen)
ament_package()
