cmake_minimum_required(VERSION 3.8)
project(frontier_probe)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
set(dependencies ament_index_cpp bspline_opt geometry_msgs nav_msgs path_searching plan_env rclcpp rosgraph_msgs scan_planner_msgs sensor_msgs std_msgs tf2 tf2_geometry_msgs tf2_ros traj_utils visualization_msgs)
foreach(dependency ${dependencies})
 find_package(${dependency} REQUIRED)
endforeach()
add_library(checked_optimizer OBJECT /workspace/src/bspline_opt/src/bspline_optimizer.cpp)
target_include_directories(checked_optimizer PRIVATE /workspace/src/bspline_opt/include)
ament_target_dependencies(checked_optimizer ${dependencies})
target_link_libraries(checked_optimizer Eigen3::Eigen)
target_compile_options(checked_optimizer PRIVATE -fsanitize=undefined -g -O1 -fno-omit-frame-pointer -D_GLIBCXX_ASSERTIONS)
add_library(native_manager OBJECT /workspace/src/plan_manage/src/scan_replan_fsm.cpp /workspace/src/plan_manage/src/planner_manager.cpp)
target_include_directories(native_manager PRIVATE /workspace/src/plan_manage/include)
ament_target_dependencies(native_manager ${dependencies})
target_link_libraries(native_manager Eigen3::Eigen)
foreach(target crash_probe)
 add_executable(${target} src/${target}.cpp src/abort_backtrace.cpp $<TARGET_OBJECTS:native_manager> $<TARGET_OBJECTS:checked_optimizer>)
 target_include_directories(${target} PRIVATE /workspace/src/plan_manage/include)
 target_compile_options(${target} PRIVATE -fno-access-control -fsanitize=undefined -g -O1 -fno-omit-frame-pointer -D_GLIBCXX_ASSERTIONS)
 target_link_options(${target} PRIVATE -fsanitize=undefined)
 ament_target_dependencies(${target} ${dependencies})
 target_link_libraries(${target} Eigen3::Eigen)
endforeach()
ament_package()
