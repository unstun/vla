cmake_minimum_required(VERSION 3.16)
project(v286_migration_export)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(scan_planner REQUIRED)
find_package(lite3_nav2_plugins REQUIRED)
find_package(lite3_nav2_interfaces REQUIRED)
find_package(nav2_core REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
add_executable(support_export support_export.cpp)
target_compile_options(support_export PRIVATE -fno-access-control)
target_include_directories(support_export BEFORE PRIVATE ${VERSIONS_ROOT})
ament_target_dependencies(support_export rclcpp scan_planner lite3_nav2_plugins lite3_nav2_interfaces nav2_core pluginlib rclcpp_lifecycle)
target_link_libraries(support_export OpenSSL::Crypto Threads::Threads)
