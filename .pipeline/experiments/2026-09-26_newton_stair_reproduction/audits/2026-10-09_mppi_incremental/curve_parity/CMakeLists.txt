cmake_minimum_required(VERSION 3.8)
project(mppi_fixed_input_curve_probe)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(plan_env REQUIRED)
find_package(scan_planner REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
add_executable(curve_probe curve_probe.cpp)
target_compile_options(curve_probe PRIVATE -fno-access-control)
target_include_directories(curve_probe PRIVATE /repo/.pipeline/experiments/2026-09-26_newton_stair_reproduction/versions)
ament_target_dependencies(curve_probe rclcpp plan_env scan_planner)
target_link_libraries(curve_probe Eigen3::Eigen OpenSSL::Crypto Threads::Threads)

option(CANDIDATE "Enable the candidate diagnostics" OFF)
if(CANDIDATE)
  target_compile_definitions(curve_probe PRIVATE DIAGNOSTIC_CANDIDATE)
