cmake_minimum_required(VERSION 3.16)
project(v177_stop_replay)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(plan_env REQUIRED)
find_package(scan_planner REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
add_executable(stop_replay stop_replay.cpp)
target_compile_options(stop_replay PRIVATE -fno-access-control)
target_include_directories(stop_replay BEFORE PRIVATE ${VERSIONS_ROOT})
ament_target_dependencies(stop_replay rclcpp plan_env scan_planner)
target_link_libraries(stop_replay OpenSSL::Crypto Threads::Threads)
