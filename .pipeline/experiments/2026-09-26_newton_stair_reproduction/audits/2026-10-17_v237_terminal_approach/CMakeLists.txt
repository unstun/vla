cmake_minimum_required(VERSION 3.16)
project(terminal_approach_probe)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(scan_planner REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
add_executable(replay_probe replay_probe.cpp)
target_compile_options(replay_probe PRIVATE -fno-access-control)
target_include_directories(replay_probe BEFORE PRIVATE ${VERSIONS_ROOT})
ament_target_dependencies(replay_probe rclcpp scan_planner)
target_link_libraries(replay_probe OpenSSL::Crypto Threads::Threads)
option(V238_CONTRACTS "Exercise candidate terminal/freshness boundaries" OFF)
if(V238_CONTRACTS)
  target_compile_definitions(replay_probe PRIVATE V238_CONTRACTS=1)
endif()
option(V240_CONTRACTS "Exercise input drain without map adoption" OFF)
if(V240_CONTRACTS)
  target_compile_definitions(replay_probe PRIVATE V240_CONTRACTS=1)
endif()
