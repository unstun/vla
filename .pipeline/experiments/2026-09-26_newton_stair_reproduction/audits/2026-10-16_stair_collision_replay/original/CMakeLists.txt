cmake_minimum_required(VERSION 3.16)
project(original_scan_collision_probe)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(message_filters REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(OpenCV REQUIRED)
find_package(PCL REQUIRED COMPONENTS common io filters)
find_package(pcl_conversions REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(visualization_msgs REQUIRED)
add_executable(original_probe original_probe.cpp ${REFERENCE}/src/grid_map.cpp ${REFERENCE}/src/raycast.cpp)
target_compile_options(original_probe PRIVATE -fno-access-control)
target_include_directories(original_probe BEFORE PRIVATE ${REFERENCE}/include ${VERSIONS_ROOT} ${PCL_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS})
ament_target_dependencies(original_probe cv_bridge geometry_msgs message_filters nav_msgs pcl_conversions rclcpp sensor_msgs tf2 tf2_ros visualization_msgs)
target_link_libraries(original_probe Eigen3::Eigen ${PCL_LIBRARIES} ${OpenCV_LIBS})
