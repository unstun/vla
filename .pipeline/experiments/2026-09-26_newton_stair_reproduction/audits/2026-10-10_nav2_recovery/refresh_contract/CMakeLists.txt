cmake_minimum_required(VERSION 3.16)
project(post_search_refresh_probe)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(plan_env REQUIRED)
find_package(scan_planner REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
add_executable(refresh_probe refresh_probe.cpp)
target_compile_options(refresh_probe PRIVATE -fno-access-control)
target_include_directories(refresh_probe BEFORE PRIVATE ${VERSIONS_ROOT})
ament_target_dependencies(refresh_probe rclcpp plan_env scan_planner)
target_link_libraries(refresh_probe OpenSSL::Crypto Threads::Threads)

set_target_properties(refresh_probe PROPERTIES ENABLE_EXPORTS ON)
