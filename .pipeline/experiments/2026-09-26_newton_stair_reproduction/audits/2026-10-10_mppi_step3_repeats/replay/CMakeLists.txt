cmake_minimum_required(VERSION 3.8)
project(recorded_validation_replay)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(plan_env REQUIRED)
find_package(scan_planner REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
option(CANDIDATE "Use v127 independent validator" OFF)
foreach(probe curve_replay map_replay)
 add_executable(${probe} ${probe}.cpp)
 target_compile_options(${probe} PRIVATE -fno-access-control)
 target_include_directories(${probe} BEFORE PRIVATE ${NATIVE_WORKSPACE}/src/scan_nav2_controller/include ${VERSIONS_ROOT})
 ament_target_dependencies(${probe} rclcpp plan_env scan_planner)
 target_link_libraries(${probe} Eigen3::Eigen OpenSSL::Crypto Threads::Threads)
 if(CANDIDATE)
  target_compile_definitions(${probe} PRIVATE CANDIDATE)
 endif()
endforeach()
